# Phase 3 — NLP / LMTA engine dependencies.
# scikit-learn is used for: TF-IDF vectorization (app/nlp/tfidf.py) and
# the TF-IDF + Logistic Regression intent classifier (app/nlp/intent.py).
# numpy is a direct dependency of scikit-learn's array operations, used
# directly in app/nlp/semantics.py for cosine similarity.

numpy==2.4.4

# Phase 3 CORRECTION — transformer embeddings (app/nlp/embeddings.py).
# Replaces the original HashingVectorizer-based embedding (which was not
# a transformer embedding and did not satisfy the spec) with a real
# sentence-transformers model: sentence-transformers/all-MiniLM-L6-v2
# (384 dimensions). This single package transitively pulls in torch and
# transformers (both required to actually run the model) — no other new
# top-level dependency was added. See app/nlp/embeddings.py's module
# docstring for the exact environmental limitation this build hit
# (huggingface.co unreachable in this sandbox) and what changes once
# that access is available.
sentence-transformers==6.1.0

# Phase 5 — SLM inference dependencies.
#
# transformers: AutoTokenizer + AutoModelForCausalLM for TinyLlama inference.
#   Already transitively pulled by sentence-transformers, pinned here for
#   explicitness. We require >=4.40 for TinyLlama chat-template support.
#
# torch: Required for model inference; already pulled by sentence-transformers.
#   Pinned here explicitly. CPU inference works without CUDA.
#
# NOTE: peft, trl, datasets, accelerate are TRAINING-ONLY dependencies.
# They are intentionally NOT listed here (requirements.txt = production
# application dependencies). Install them separately for training:
#   pip install peft trl datasets accelerate
#
# The application itself (FastAPI + inference) works with ONLY the packages
# listed below — training libraries are never imported at application startup.
transformers>=4.40.0
torch>=2.0.0
